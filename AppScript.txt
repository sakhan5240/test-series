function doPost(e) {
  let response = { success: false, message: "Unknown error occurred" };
  
  if (!e || !e.postData || !e.postData.contents) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Invalid Request." }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  try {
    let requestData = JSON.parse(e.postData.contents);
    let action = requestData.action;
    let ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // --- 1. LOGIN SYSTEM ---
    if (action === "login") {
      let sheet = ss.getSheetByName("Users");
      let data = sheet.getDataRange().getValues();
      let username = String(requestData.username || "").trim();
      let password = String(requestData.password || "").trim();
      
      let isValid = false;
      for (let i = 1; i < data.length; i++) {
        if (!data[i][0]) continue; 
        if (String(data[i][0]).trim() === username && String(data[i][1]).trim() === password) {
          isValid = true; break;
        }
      }
      if (isValid) response = { success: true, message: "Login successful!" };
      else response = { success: false, message: "Invalid credentials." };
    }
    
    // --- 2. GET DASHBOARD & DATES ---
    else if (action === "getDashboard") {
      let username = String(requestData.username || "").trim();
      let testsSheet = ss.getSheetByName("Tests");
      let resultsSheet = ss.getSheetByName("Results");
      
      let testData = testsSheet.getDataRange().getValues();
      let activeTests = [];
      
      for (let i = 1; i < testData.length; i++) {
        if (!testData[i][0]) continue; 
        let isActive = testData[i][3] === true || String(testData[i][3]).toUpperCase() === "TRUE";
        
        if (isActive) {
          // SAFETY FIX: Safely parse Published Date (Column E / Index 4)
          let pubDateRaw = (testData[i].length > 4 && testData[i][4] !== "") ? testData[i][4] : null;
          let pubDateISO = null;
          if (pubDateRaw) {
            try { pubDateISO = new Date(pubDateRaw).toISOString(); } catch(err) { pubDateISO = null; }
          }

          activeTests.push({
            testId: String(testData[i][0]).trim(),
            title: String(testData[i][1] || "Untitled Test"),
            duration: parseInt(testData[i][2]) || 60,
            publishedDate: pubDateISO, // Safe Date Sent
            status: "pending" 
          });
        }
      }
      
      let pastAnalysis = {};
      if (resultsSheet) {
        let resultsData = resultsSheet.getDataRange().getValues();
        for (let i = 1; i < resultsData.length; i++) {
          if (!resultsData[i][1]) continue; 
          if (String(resultsData[i][1]).trim() === username) {
            let tId = String(resultsData[i][2]).trim();
            let testIndex = activeTests.findIndex(t => t.testId === tId);
            if (testIndex !== -1) {
              activeTests[testIndex].status = "completed";
              activeTests[testIndex].score = resultsData[i][3];
              activeTests[testIndex].total = resultsData[i][4];
            }
            try {
              if (resultsData[i][6]) pastAnalysis[tId] = JSON.parse(resultsData[i][6]); 
            } catch (jsonError) { pastAnalysis[tId] = { error: "Unavailable" }; }
          }
        }
      }
      response = { success: true, dashboard: activeTests, history: pastAnalysis };
    }
    
    // --- 3. FETCH TEST QUESTIONS ---
    else if (action === "fetchTest") {
      let testName = String(requestData.testName || "").trim();
      let sheet = ss.getSheetByName(testName);
      
      if (!sheet) {
        response = { success: false, message: "Questions not found." };
      } else {
        let data = sheet.getDataRange().getValues();
        let questions = [];
        for (let i = 1; i < data.length; i++) {
          if (data[i][0]) { 
            let cleanOptions = [data[i][1], data[i][2], data[i][3], data[i][4]].filter(opt => opt !== "" && opt !== null && opt !== undefined);
            questions.push({ id: i, questionText: String(data[i][0]), options: cleanOptions });
          }
        }
        response = { success: true, questions: questions };
      }
    }
    
    // --- 4. SECURE SUBMIT TEST ---
    else if (action === "submitTest") {
      let username = String(requestData.username || "").trim();
      let testName = String(requestData.testName || "").trim();
      let userAnswers = requestData.answers || {}; 
      
      let testSheet = ss.getSheetByName(testName);
      let testData = testSheet.getDataRange().getValues();
      let score = 0, totalQuestions = 0; 
      let analysisDetails = [];
      
      for (let i = 1; i < testData.length; i++) {
        if (testData[i][0]) { 
          totalQuestions++;
          let qId = i;
          let questionText = String(testData[i][0]);
          let correctAnswer = testData[i].length > 5 ? String(testData[i][5]).trim() : ""; 
          let studentAnswer = userAnswers[qId] ? String(userAnswers[qId]).trim() : "Unanswered";
          
          let isCorrect = (studentAnswer !== "Unanswered" && studentAnswer === correctAnswer);
          if (isCorrect) score++;
          
          analysisDetails.push({ questionText, studentAnswer, correctAnswer, isCorrect });
        }
      }
      
      let percentage = totalQuestions > 0 ? ((score / totalQuestions) * 100).toFixed(1) : 0;
      
      let lock = LockService.getScriptLock();
      lock.waitLock(15000); 
      try {
        let resultsSheet = ss.getSheetByName("Results");
        if (!resultsSheet) {
          resultsSheet = ss.insertSheet("Results");
          resultsSheet.appendRow(["Timestamp", "Username", "TestID", "Score", "Total", "Percentage", "AnalysisJSON"]);
        }
        resultsSheet.appendRow([
          new Date(), username, testName, score, totalQuestions, percentage + "%", 
          JSON.stringify({ score, total: totalQuestions, percentage, details: analysisDetails })
        ]);
        SpreadsheetApp.flush(); 
      } finally { lock.releaseLock(); }
      
      response = { success: true, score, total: totalQuestions, percentage, analysis: analysisDetails };
    }
    
  } catch (error) { response = { success: false, message: "Backend Sync Error: " + error.message }; }
  
  return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
}

function doOptions(e) { return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.JSON); }
